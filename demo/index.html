🌱 esg-badge

Zero-dependency ESG score badge & widget for any website.

Default (light)
Dark theme
Compact
Low ESG (risk)

Usage

<script src="esg-badge.js"></script>
<div id="esg-badge"></div>
<script>
ESGBadge.create({
  container: '#esg-badge',
  company: 'Hoju LLC',
  scores: { E: 82, S: 76, G: 90 },
  theme: 'light',   // or 'dark'
  updated: 'Q3 2026',
});
</script>
